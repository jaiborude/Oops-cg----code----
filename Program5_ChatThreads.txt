class MyChatRoom {
    private boolean active = true;

    // Synchronized method to send a message
    public synchronized void sendMessage(String user, String message) {
        System.out.println(user + ": " + message);
        try {
            Thread.sleep(500); // Simulate delay
        } catch (InterruptedException e) {
            System.out.println(user + " was interrupted.");
        }
    }

    public boolean isActive() {
        return active;
    }

    public void stopChat() {
        active = false;
    }
}

class User extends Thread {
    private String username;
    private MyChatRoom chatRoom;
    private String[] messages;
    private boolean suspended = false;

    public User(String name, MyChatRoom chatRoom, String[] messages) {
        this.username = name;
        this.chatRoom = chatRoom;
        this.messages = messages;
    }

    public void run() {
        /*if(username.equals("Anand"))
        {
            System.out.println(username+" Has Created the Chatroom");
            //System.out.println(username+" Welcomes you to his chatroom");
        }*/
        for (String message : messages) {
            synchronized (this) {
                while (suspended) {
                    try {
                        wait();
                    } catch (InterruptedException e) {
                        System.out.println(username + " was interrupted.");
                    }
                }
            }
            if (!chatRoom.isActive()) break;
            chatRoom.sendMessage(username, message);
        }
        System.out.println(username + " has left the chat.");
    }
}

public class MyChatSimulation {
    public static void main(String[] args) {
        System.out.println("Welcom to Chating Application");
        MyChatRoom chatRoom = new MyChatRoom();

        String[] user1Msgs = {"Hi!", "How are you?", "That's great!"};
        String[] user2Msgs = {"Hello!", "I'm fine.", "Let's meet."};
        String[] user3Msgs = {"Hello!", "I'm Good.", "Yes Sure let's meet tomorrow."};

        User user1 = new User("Anand", chatRoom, user1Msgs);
        User user2 = new User("Kiran", chatRoom, user2Msgs);
        User user3 = new User("Ganesh", chatRoom, user3Msgs);

        user1.setPriority(Thread.MAX_PRIORITY);
        user2.setPriority(Thread.MIN_PRIORITY);
        user3.setPriority(Thread.NORM_PRIORITY);

        user1.start();
        user2.start();
        user3.start();

        // Wait for user2 to finish first
        try {
            if (user2.isAlive()) {
                user2.join();
            }

            // Suspend and resume user3
            System.out.println("Suspending Ganesh...");
            //user1.suspendUser();
            user3.suspend();
            Thread.sleep(1500); // simulate pause
            System.out.println("Resuming Ganesh...");
            user3.resume();
            user3.join();
        } catch (InterruptedException e) {
            System.out.println("Main thread interrupted.");
        }

        chatRoom.stopChat();
        System.out.println("Chat ended.");
    }
}
