import java.util.InputMismatchException;
import java.util.Scanner;

public class MyCalculator {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        boolean continueCalculation = true;
        System.out.println("Welcome to the Java Calculator!");

        while (continueCalculation) {
            double num1 = 0;
            double num2 = 0;
            char operator = ' ';
            boolean validInput = false;

            // Get first number
            while (!validInput) {
                try {
                    System.out.print("Enter the first number: ");
                    num1 = scanner.nextDouble();
                    validInput = true;
                } catch (InputMismatchException e) {
                    System.out.println("Invalid input. Please enter a valid number.");
                    scanner.next(); // Consume the invalid input
                }
            }
            validInput = false; // Reset for next input

            // Get operator
            while (!validInput) {
                System.out.print("Choose an operator (+, -, *, /, %): ");
                String opInput = scanner.next();
                if (opInput.length() == 1 && "+-*/%".contains(opInput)) {
                    operator = opInput.charAt(0);
                    validInput = true;
                } else {
                    System.out.println("Invalid operator. Please choose from +, -, *, /, %.");
                }
            }
            validInput = false; // Reset for next input

            // Get second number
            while (!validInput) {
                try {
                    System.out.print("Enter the second number: ");
                    num2 = scanner.nextDouble();
                    if (operator == '/' && num2 == 0) {
                        System.out.println("Error: Division by zero is not allowed.");
                    } else {
                        validInput = true;
                    }
                } catch (InputMismatchException e) {
                    System.out.println("Invalid input. Please enter a valid number.");
                    scanner.next(); // Consume the invalid input
                }
            }

            double result = 0;
            switch (operator) {
                case '+':
                    result = num1 + num2;
                    break;
                case '-':
                    result = num1 - num2;
                    break;
                case '*':
                    result = num1 * num2;
                    break;
                case '/':
                    result = num1 / num2;
                    break;
                case '%':
                    result = num1 % num2;
                    break;
                default:
                    // This case should ideally not be reached due to input validation
                    System.out.println("An unexpected error occurred with the operator.");
            }

            System.out.println("Result: " + num1 + " " + operator + " " + num2 + " = " + result);
            System.out.print("Do you Wish to Perform another Calculation? (yes/no): ");
            String choice = scanner.next().toLowerCase();
            if (!choice.equals("yes")) {
                continueCalculation = false;
            }
        }
        System.out.println("Thank you for using the calculator!");
        scanner.close();
    }
}
